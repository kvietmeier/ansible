###======== Upgrade RHEL/Centos Based Systems =========###
- name: Install the RHEL/Centos Development Tools package group
  yum: 
    name: "@Development tools"
    state: present
  when: ansible_distribution == 'CentOS' or ansible_distribution == 'Red Hat Enterprise Linux'
  tags:
     - pkgs

- name: Install a Bunch of Useful Packages you will need eventually
  when: ansible_distribution == 'CentOS' or ansible_distribution == 'Red Hat Enterprise Linux'
  yum: 
    name: '{{ xtra_pkgs }}'
    state: present
  tags:
    - pkgs

- name: RHEL - Install RHEL Specific Useful Packages
  when: ansible_distribution == 'CentOS' or ansible_distribution == 'Red Hat Enterprise Linux'
  yum:
    name: '{{ rhel_pkgs }}'
    state: present
  tags:
     - pkgs

- name: "Disable epel until you really need it"
  when: ansible_distribution == 'CentOS' or ansible_distribution == 'Red Hat Enterprise Linux'
  command: sed -i -e "s/^enabled=1/enabled=0/" /etc/yum.repos.d/epel.repo
  tags:
     - pkgs

- name: "RHEL: Upgrade all packages"
  when: ansible_distribution == 'CentOS' or ansible_distribution == 'Red Hat Enterprise Linux'
  yum:
    name: "*" 
    state: latest
  tags:
     - pkgs


###======== Upgrade Debian/Ubuntu based systems and reboot if necessary ========###
- name: "Ubuntu: Check if there are packages available to be installed/upgraded"
  when: ansible_distribution == 'Ubuntu'
  command: /usr/lib/update-notifier/apt-check --package-names

- name: "Update apt cache if it is old"
  when: (ansible_distribution == 'Ubuntu')
  apt:
    update_cache: yes

    cache_valid_time: 3600
  tags:
     - pkgs

- name: "Ubuntu: Upgrade all packages to the latest version"
  when: (ansible_distribution == 'Ubuntu') and (packages.stderr != "")
  apt: 
    name: "*"
    state: latest
  tags:
    - pkgs

- name: "Ubuntu: Install Ubuntu Development Tools"
  when: ansible_distribution == 'Ubuntu'
  apt:
    name: build-essential
    state: present
  tags: 
    - pkgs

# Check for the binaries. We only want to compile if the 
# binaries don't already exist - this isn't an "idempotent"
# process so if we don't check these tasks execute every time.
- name: iperf3 present?
  stat:
    path: "{{ iperf3_binary }}"
  register: iperf3

- name: FIO present?
  stat:
    path: "{{ fio_binary }}"
  register: fio

- name: Sockperf present?
  stat:
    path: "{{ sperf_binary }}"
  register: sockperf

###==============================  Compile Tools  ==============================###
###---- Compile FIO
- name: Clone FIO from git
  git: 
    repo: "{{ fio_repo }}"
    dest: "{{ git_dir }}/fio" 
    clone: yes
    force: yes
  when: fio.stat.exists == false
  tags:
    - fio

- name: Compile FIO
  command: "{{ item }} chdir={{ fio_dir }}"
  with_items:
   - ./configure
   - /usr/bin/make
   - /usr/bin/make install
  when: fio.stat.exists == false
  tags:
    - fio  

###---- Compile iperf3
- name: Clone iperf3 from git
  git: 
    repo: "{{ iperf3_repo }}"
    dest: "{{ git_dir }}/iperf3" 
    clone: yes
    force: yes
  when: iperf3.stat.exists == false
  tags:
    - iperf3

- name: Compile iperf3
  command: "{{ item }} chdir={{ iperf3_dir }}"
  with_items:
   - ./configure
   - /usr/bin/make
   - /usr/bin/make install
   - /usr/sbin/ldconfig
  when: iperf3.stat.exists == false
  tags:
    - iperf3  
 
###---- Compile sockperf
- name: Clone sockperf from git
  git: 
    repo: "{{ sockperf_repo }}"
    dest: "{{ git_dir }}/sockperf" 
    clone: yes
    force: yes
  when: sockperf.stat.exists == false
  tags:
    - sockperf

- name: Compile sockperf Step 1
  command: ./autogen.sh
  args:
    chdir: "{{ sockperf_dir }}"
  when: sockperf.stat.exists == false
  tags:
    - sockperf  

- name: Compile sockperf Step 2
  command: ./configure --prefix= --enable-test
  args:
    chdir: "{{ sockperf_dir }}"
  when: sockperf.stat.exists == false
  tags:
    - sockperf  

- name: Compile sockperf Step 3
  command: make
  args:
    chdir: "{{ sockperf_dir }}"
  when: sockperf.stat.exists == false
  tags:
    - sockperf  

- name: Compile sockperf Step 4
  command: make install
  args:
    chdir: "{{ sockperf_dir }}"
  when: sockperf.stat.exists == false
  tags:
    - sockperf

# Setup some basic functionality for users and sudo

#--- Remove the requiretty directive from /etc/sudoers
- name: Remove requiretty
  lineinfile:
    dest: /etc/sudoers
    regexp: "^Defaults.*requiretty"
    line: 'Defaults !requiretty'
    backup: yes
    state: present
    validate: 'visudo -cf %s'
  become: yes
  become_method: sudo
  tags:
     - usercfg

- name: Enable wheel group in Centos/RHEL to issue all commands with NOPASSWD
  when: ansible_distribution == 'CentOS' or ansible_distribution == 'Red Hat Enterprise Linux'
  lineinfile:
    dest: /etc/sudoers
    state: present
    backup: yes
    regexp: '^%wheel.*NOPASSWD'
    line: '%wheel ALL=(ALL) NOPASSWD: ALL'
    validate: 'visudo -cf %s'
  become: yes
  become_method: sudo
  tags:
     - usercfg
