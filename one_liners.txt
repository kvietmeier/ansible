# =============================================================================
# 1. LEARN — Ansible ad-hoc patterns
#    Form:  ansible -i INVENTORY HOSTS [-b] [-m MODULE] [-a ARGS]
#    -a alone uses the "command" module (no shell metacharacters).
# =============================================================================

# --- connectivity / facts ---
ansible -i ./inventory.ini clients -m ping
ansible -i ./inventory.ini clients -a date
ansible -i ./inventory.ini clients -a "df -kh"
ansible -i ./inventory.ini clients -m setup -a "filter=ansible_distribution*"
ansible -i ./inventory.ini clients -m shell -a "uname -r; free -h | head -2"

# --- limit / parallelism ---
ansible -i ./inventory.ini clients -l client01 -a hostname
ansible -i ./inventory.ini clients -f 10 -a uptime

# --- packages (become) ---
ansible -i ./inventory.ini clients -b -m apt -a "name=nfs-common state=present update_cache=yes"
# RHEL-family example:
# ansible -i ./inventory.ini clients -b -m dnf -a "name=nfs-utils state=present"

# --- files / copy ---
ansible -i ./inventory.ini clients -b -m file -a "path=/mount/vast/fio state=directory mode=0777"
ansible -i ./inventory.ini clients -m copy -a "src=./README.md dest=/tmp/README.md mode=0644"

# --- shell vs command ---
# command (-a): no pipes/redirects; shell: full shell
ansible -i ./inventory.ini clients -a "ls /mount/vast"
ansible -i ./inventory.ini clients -m shell -a "ls -la /mount/vast | head"

# --- services / checks ---
ansible -i ./inventory.ini clients -b -m service -a "name=rpcbind state=started enabled=yes"
ansible -i ./inventory.ini clients -a "command -v fio; command -v elbencho; fio --version 2>/dev/null | head -1"

# =============================================================================
# 2. LAB — playbooks (client polish / platform wiring)
# =============================================================================

ansible-playbook -i ./inventory.ini site.yml --tags client
ansible-playbook -i ./inventory.ini site.yml --tags git_tools
ansible-playbook -i ./inventory.ini site.yml --tags install_driver
ansible-playbook -i ./inventory.ini site.yml --tags mountprep
ansible-playbook -i ./inventory.ini site.yml --tags mount_now \
  -e "mount_point=/mount/vast view_path=nfs_01 dns_alias=sharespool dns_domain=busab"

# =============================================================================
# 3. LAB — elbencho multi-client benchmark examples
# =============================================================================

ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags mkdirs
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags mount
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags elbencho_serv
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags copy_scripts
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags kill_all
