# =============================================================================
# 1. CURRENT — inventory / ad-hoc
# =============================================================================

# Date / connectivity
ansible -i ./inventory.ini clients -a date
ansible -i ./inventory.ini clients -a "df -kh"

# =============================================================================
# 2. CURRENT — playbooks (swiss-army client polish)
# =============================================================================

# Day-2 client polish (users, /mount/vast dirs, ~/tools clones)
ansible-playbook -i ./inventory.ini site.yml --tags client

# Tools clone only
ansible-playbook -i ./inventory.ini site.yml --tags git_tools

# VAST NFS driver
ansible-playbook -i ./inventory.ini site.yml --tags install_driver

# Mount prep / mount now (override VIP/view as needed)
ansible-playbook -i ./inventory.ini site.yml --tags mountprep
ansible-playbook -i ./inventory.ini site.yml --tags mount_now \
  -e "mount_point=/mount/vast view_path=nfs_01 dns_alias=sharespool dns_domain=busab"

# =============================================================================
# 3. CURRENT — elbencho campaigns (project / multi-client)
# =============================================================================

ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags mkdirs
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags mount
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags elbencho_serv
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags copy_scripts
ansible-playbook -i ./inventory.ini playbooks/elbencho.yml --tags kill_all

# Manual on primary after copy_scripts:
#   ~/elbencho/elbencho_nfs_prep.sh <NUM_CLIENTS>
#   ~/elbencho/elbencho_nfs_testblk_sizes.sh <NUM_CLIENTS>
